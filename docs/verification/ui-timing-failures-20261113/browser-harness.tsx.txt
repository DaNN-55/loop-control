import {useState} from 'react';
import {createRoot} from 'react-dom/client';
import {LocalDeletionRecovery} from './src/local-app/LocalDeletionRecovery';
import {WorkspaceNarrationPanel} from './src/local-app/WorkspaceNarrationPanel';
import {WorkspaceDraftOwner} from './src/local-app/WorkspaceDraft';
import {WorkspaceMaterialSelectionProvider,workspaceMaterialDragType} from './src/local-app/WorkspaceMaterialDrop';
import {localCommand} from './src/local-app/client';
import './src/styles.css';
const trace:any={requests:[],confirmKeys:[],readbackKeys:[],saveKeys:[],focus:[]};
let releaseList!:()=>void,releaseWorkspace!:()=>void,releaseReceipt!:()=>void;
const listGate=new Promise<void>(r=>{releaseList=r;}),workspaceGate=new Promise<void>(r=>{releaseWorkspace=r;}),receiptGate=new Promise<void>(r=>{releaseReceipt=r;});
let listReleased=false,holdWorkspace=false;
const prepared={cleanupKey:'prepared-key',title:'未提交的生产单',path:'/synthetic/order-one',phase:'prepared'},committed={cleanupKey:'committed-key',title:'文件待清理的生产单',path:'/synthetic/order-two'};
let stored:any={id:'w',episodeId:'e',revision:1,stale:false,project:{sha256:'saved-sha'},clips:[{id:'clip',shotId:'S01',itemId:'item'}],candidates:[],narration:[{clipId:'clip',shotId:'S01',mode:'manual',enabled:true,text:'镜头正文',status:'pending',inputFingerprint:'initial'}]};
const receipts=new Map<string,unknown>();let lost=false;
window.fetch=async(input,init)=>{
 const path=String(input);trace.requests.push({path,method:init?.method??'GET'});
 if(path.endsWith('/session'))return Response.json({csrfToken:'synthetic-browser-token'});
 if(path.endsWith('/deletion-cleanups')){if(!listReleased)await listGate;return Response.json([prepared,committed]);}
 if(init?.method==='POST'){
  const body=JSON.parse(String(init.body));
  if(path.endsWith('/command-receipt')){trace.readbackKeys.push(body.operation.idempotencyKey);if(trace.readbackKeys.length===1)return Response.json({status:'unknown'});await receiptGate;return Response.json({status:'committed',result:receipts.get(body.operation.idempotencyKey)});}
  if(body.command==='confirmWorkspaceNarration')trace.confirmKeys.push(body.idempotencyKey);
  if(body.command==='saveWorkspaceNarration')trace.saveKeys.push(body.idempotencyKey);
  if(receipts.has(body.idempotencyKey))return Response.json(receipts.get(body.idempotencyKey));
  if(body.command==='saveWorkspaceNarration')stored={...stored,revision:2,narration:[{...stored.narration[0],audioTrackId:'track-first',inputFingerprint:'track-first'}],candidates:[{id:'track-first',audioTrackId:'track-first',kind:'audio',shotId:'S01',relativePath:'voice/first.mp3',text:'镜头正文',durationSeconds:2,status:'available'}]};
  else if(body.command==='confirmWorkspaceNarration')stored={...stored,revision:3};else throw Error('Unexpected mutation: '+body.command);
  receipts.set(body.idempotencyKey,stored);if(body.command==='confirmWorkspaceNarration'&&!lost){lost=true;return Response.json({error:'确认回包中断'},{status:503});}return Response.json(stored);
 }
 if(path.endsWith('/workspace')){if(holdWorkspace)await workspaceGate;return Response.json({workspace:stored});}
 return Response.json({serviceCatalog:{version:1,groups:[]}});
};
document.addEventListener('focusin',()=>trace.focus.push({key:(document.activeElement as HTMLElement)?.dataset.cleanupKey??'',label:document.activeElement?.textContent?.slice(0,70)}));
(window as any).timing={trace,releaseList:()=>{listReleased=true;releaseList();},releaseWorkspace,releaseReceipt,hold:()=>{holdWorkspace=true;},drop:()=>{const target=document.querySelector<HTMLButtonElement>('button[aria-label="拖入口播文件"]');if(!target)throw Error('Missing public drop target');const data=new DataTransfer();data.setData(workspaceMaterialDragType,'first');target.dispatchEvent(new DragEvent('drop',{bubbles:true,dataTransfer:data}));}};
function FocusCase(){const[revision,setRevision]=useState(1),[key,setKey]=useState(committed.cleanupKey);return <><h1>清理任务焦点验收</h1><button onClick={()=>(window as any).timing.releaseList()}>返回清理列表</button><button onClick={()=>setRevision(r=>r+1)}>重新读取</button><button onClick={()=>setKey(prepared.cleanupKey)}>定位其他任务</button><LocalDeletionRecovery revision={revision} targetKey={key}/></>;}
const materials=[{id:'first',source_path:'first.mp3',storage_path:'voice/first.mp3',material_type:'audio',material_purpose:'general_reference'}];
function NarrationView(){const[w,setW]=useState(stored);return <WorkspaceDraftOwner ownerId="browser-ui-timing-owner"><WorkspaceMaterialSelectionProvider materials={materials}><WorkspaceNarrationPanel episodeId="e" workspace={w} selectedShotId="S01" pending={false} command={(name,data)=>localCommand(name,data,undefined,'episode-commands')} onChanged={setW} materials={materials}/></WorkspaceMaterialSelectionProvider></WorkspaceDraftOwner>;}
function ManualCase(){const[open,setOpen]=useState(true);return <><h1>未知确认回执验收</h1><button onClick={()=>{setOpen(false);(window as any).timing.hold();}}>关闭口播测试</button><button onClick={()=>setOpen(true)}>重开口播测试</button><button onClick={()=>releaseWorkspace()}>返回工程读取</button><button onClick={()=>releaseReceipt()}>返回确认回执</button>{open&&<NarrationView/>}</>;}
createRoot(document.getElementById('root')!).render(location.search.includes('manual')?<ManualCase/>:<FocusCase/>);
