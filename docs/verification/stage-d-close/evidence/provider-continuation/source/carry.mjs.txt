import assert from'node:assert/strict';import{readFileSync,writeFileSync,mkdirSync,existsSync,openSync,closeSync,fsyncSync,renameSync,unlinkSync}from'node:fs';import{join}from'node:path';import{createHash,randomUUID}from'node:crypto';
export const hash=p=>createHash('sha256').update(readFileSync(p)).digest('hex');export const json=p=>JSON.parse(readFileSync(p));export const save=(p,v)=>writeFileSync(p,JSON.stringify(v,null,2)+'\n',{flag:'wx',mode:0o600});
export function carry(parent,directory,provider,approvalFile){
 assert(['pexels','google_tts','volcengine_tts'].includes(provider));assert(!existsSync(parent+'/request.lock'),'Parent lock must not be taken over');const p=json(parent+'/policy.json'),s=json(parent+'/state.json');assert.equal(hash(parent+'/policy.json'),s.policySha256);assert.equal(s.halted,true,'Prior phase must be sealed');assert(s.events.at(-1)?.outcome!=='reserved');assert(s.used<=17&&s.costMicrounits<=2000000);mkdirSync(directory,{mode:0o700});const operations=structuredClone(p.operations);for(const[k,v]of Object.entries(operations))if(['freesound','cloudflare','openai'].some(x=>k.startsWith(x+'.')))v.maxRequests=s.byOperation[k]??0;
 const parentProof={directory:parent,policySha256:hash(parent+'/policy.json'),stateSha256:hash(parent+'/state.json'),used:s.used,costMicrounits:s.costMicrounits},policy={...p,runId:randomUUID(),enabled:true,expiresAt:new Date(Date.now()+3600000).toISOString(),maxRequests:17,currency:'CNY',maxCostMicrounits:2000000,approvalSha256:hash(approvalFile),providers:[provider],operations,continuation:parentProof};const bytes=JSON.stringify(policy,null,2)+'\n',policySha256=createHash('sha256').update(bytes).digest('hex');writeFileSync(directory+'/policy.json',bytes,{flag:'wx',mode:0o600});save(directory+'/state.json',{...s,policySha256,halted:false});save(directory+'/parent-proof.json',parentProof);save(directory+'/parent-state.json',s);return{directory,policySha256};
}
export function seal(directory,reason){
 const lock=directory+'/request.lock',fd=openSync(lock,'wx',0o600);try{const s=json(directory+'/state.json');assert(s.events.at(-1)?.outcome!=='reserved','Uncertain in-flight request is not refundable');const before=structuredClone(s);s.halted=true;const temp=directory+'/'+randomUUID()+'.tmp',f=openSync(temp,'wx',0o600);try{writeFileSync(f,JSON.stringify(s,null,2)+'\n');fsyncSync(f);}finally{closeSync(f);}renameSync(temp,directory+'/state.json');const dir=openSync(directory,'r');try{fsyncSync(dir);}finally{closeSync(dir);}save(directory+'/seal.json',{reason,used:s.used,costMicrounits:s.costMicrounits,eventsUnchanged:JSON.stringify(before.events)===JSON.stringify(s.events),countsUnchanged:JSON.stringify(before.byOperation)===JSON.stringify(s.byOperation)});return s;}finally{closeSync(fd);unlinkSync(lock);}
}
