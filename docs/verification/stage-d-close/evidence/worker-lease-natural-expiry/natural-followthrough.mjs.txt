// Explicitly cover the legitimate running -> ready(backoff) -> running -> completed sequence.
import assert from'node:assert/strict';import{appendFileSync,readFileSync}from'node:fs';import{performance}from'node:perf_hooks';import{run,load,save,taskState,sleep,login,verifyCompleted}from'./control.mjs';const f=load('natural-watch-private.json'),initial=f.initial,start=performance.now(),samples=[];let firstReady,firstNew;
while(performance.now()-start<1200000){const t=taskState(f.episode.id),s={dbNow:t.dbNow,status:t.status,attempt:t.attempt,claimedAt:t.claimedAt,oldLeaseElapsedSeconds:(Date.parse(t.dbNow)-Date.parse(initial.claimedAt))/1000,observerMonotonicMs:performance.now()-start};appendFileSync(run+'natural-followthrough-observations.jsonl',JSON.stringify(s)+'\n',{mode:0o600});samples.push(s);if(t.attempt===1&&t.status==='running'){assert.equal(t.claimedAt,initial.claimedAt);assert.equal(t.token,initial.token);}else if(t.attempt===1&&t.status==='ready'){assert(s.oldLeaseElapsedSeconds>=1800);assert.equal(t.token,null);assert.deepEqual(t.runs.map(r=>({attempt:r.attempt,status:r.status})),[{attempt:0,status:'failed'}]);firstReady??=s;}else{assert(s.oldLeaseElapsedSeconds>=1800);assert.equal(t.attempt,2);firstNew=s;break;}await sleep(s.oldLeaseElapsedSeconds>1770?1000:10000);}
assert(firstNew);const completed=await verifyCompleted(await login(),f.episode);assert.equal(completed.task.attempt,2);assert.deepEqual(completed.task.runs.map(r=>({attempt:r.attempt,status:r.status})),[{attempt:0,status:'failed'},{attempt:1,status:'completed'}]);assert.notEqual(completed.task.runs[0].token,completed.task.runs[1].token);save('natural-completed-private.json',completed);save('natural-recovery-result.json',{passed:true,taskId:initial.id,claimedAt:initial.claimedAt,lastObservedStillOriginal:samples.findLast(s=>s.attempt===1&&s.status==='running'),firstObservedReady:firstReady??null,firstObservedNewAttempt:firstNew,sampleCount:samples.length,actualDatabaseSecondsFromLastHeartbeat:firstNew.oldLeaseElapsedSeconds,observerMonotonicMs:performance.now()-start,noManualClaimedAtChange:true,noLeaseShortening:true,noClockChange:true,originalTaskResubmittedAfterCrash:false,targetedWakeOfOriginalTaskAfterCrash:false,unrelatedEarlyHttpStarts:2,realNativeCli:true,normalApplicationFactory:true,attempts:completed.task.runs.map(r=>({attempt:r.attempt,status:r.status})),artifactShaVerified:true,artifactCount:completed.artifacts.length,earlierObserverRunningOnlyAssumptionNotAcceptanceCondition:true,onlineRequests:0});console.log('Natural recovery including ready/backoff phase passed.');
