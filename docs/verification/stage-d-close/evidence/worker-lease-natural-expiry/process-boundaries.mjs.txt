import assert from'node:assert/strict';import{writeFileSync,unlinkSync,readFileSync}from'node:fs';import{run,namespace,load,save,launch,login,post,fixture,until,events,taskState,killOwned,alive,sleep,verifyCompleted}from'./control.mjs';
const mode=process.argv[2],label=process.argv[3]??'01';assert(['worker-exit','parent-exit','graceful-stop'].includes(mode));let servicePid=load(namespace+'service-private.json').pid;const headers=await login(),e=await fixture(headers,mode);writeFileSync(run+namespace+'gate-all','controlled local pause',{mode:0o600});await post(headers,'start',{episodeId:e.id});const t=await until(()=>taskState(e.id));await until(()=>readFileSync(run+'fixture-events-private.jsonl','utf8').includes(t.id));const w=events().find(v=>v.name==='worker-started'&&v.taskId===t.id);assert(w);const before=taskState(e.id);save(mode+'-'+label+'-before-private.json',{episode:e,task:before,servicePid,workerPid:w.pid});
if(mode==='worker-exit'){killOwned(w.pid,true);unlinkSync(run+namespace+'gate-all');const completed=await verifyCompleted(headers,e);assert.equal(completed.task.attempt,2);assert.deepEqual(completed.task.runs.map(r=>r.status),['failed','completed']);save(mode+'-'+label+'-result.json',{passed:true,realWorkerGroupKilled:true,parentSurvived:alive(servicePid),failedAttemptRetained:true,attempts:completed.task.runs.map(r=>({attempt:r.attempt,status:r.status})),fencedFailureReleasedLease:true,noLeaseTimestampEdit:true,httpArtifactShaVerified:true});}
if(mode==='parent-exit'){killOwned(servicePid);await until(()=>!alive(servicePid));assert(alive(w.pid));servicePid=await launch(mode+'-'+label+'-after');await sleep(6000);const active=taskState(e.id);assert.equal(active.attempt,1);assert.equal(active.status,'running');assert.equal(active.token,before.token);assert(alive(w.pid));unlinkSync(run+namespace+'gate-all');const completed=await verifyCompleted(await login(),e);assert.equal(completed.task.attempt,1);save(mode+'-'+label+'-result.json',{passed:true,parentKilled:true,detachedWorkerSurvived:true,newServiceDidNotRevokeValidLease:true,sameAttemptCompleted:true,attempt:1,runs:completed.task.runs.length,httpArtifactShaVerified:true,notImmediateFailoverEvidence:true});}
if(mode==='graceful-stop'){process.kill(servicePid,'SIGTERM');await until(()=>!alive(servicePid)&&!alive(w.pid));const interrupted=taskState(e.id);assert.equal(interrupted.status,'ready');assert.equal(interrupted.attempt,1);assert.equal(interrupted.token,null);unlinkSync(run+namespace+'gate-all');servicePid=await launch(mode+'-'+label+'-after');const completed=await verifyCompleted(await login(),e);assert.equal(completed.task.attempt,2);save(mode+'-'+label+'-result.json',{passed:true,normalStopKilledOwnedWorkerGroup:true,readyAfterStop:true,failedAttemptRetained:true,attempts:completed.task.runs.map(r=>({attempt:r.attempt,status:r.status})),httpArtifactShaVerified:true,claimedAtNeverManuallyChanged:true});}
console.log(mode+' boundary passed');
