import assert from'node:assert/strict';import{writeFileSync,unlinkSync,readFileSync}from'node:fs';import{run,save,launch,login,post,fixture,until,events,taskState,killOwned,alive,sleep}from'./control.mjs';
const servicePid=await launch('natural-before'),headers=await login(),e=await fixture(headers,'natural expiry');writeFileSync(run+'gate-all','hold before result',{mode:0o600});await post(headers,'start',{episodeId:e.id});let t=await until(()=>taskState(e.id));await until(()=>{try{return readFileSync(run+'fixture-events-private.jsonl','utf8').includes(t.id);}catch{return false;}});t=taskState(e.id);assert.equal(t.status,'running');assert.equal(t.attempt,1);const worker=events().find(v=>v.name==='worker-started'&&v.taskId===t.id&&v.servicePid===servicePid);assert(worker);save('natural-before-kill-private.json',{episode:e,task:t,headers,servicePid,workerPid:worker.pid});
killOwned(servicePid);killOwned(worker.pid,true);await until(()=>!alive(servicePid)&&!alive(worker.pid));unlinkSync(run+'gate-all');const after=taskState(e.id);assert.equal(after.status,'running');assert.equal(after.attempt,1);assert.equal(after.claimedAt,t.claimedAt);save('natural-crash-result.json',{passed:true,episodeId:e.id,taskId:t.id,oldServiceAndWorkerKilled:true,claimedAt:t.claimedAt,observedDatabaseTime:after.dbNow,leaseMinutes:30,claimedAtManuallyChanged:false,leaseShortened:false,clockChanged:false,oldAttempt:0,oldRunStillRunning:true});const secondPid=await launch('natural-after');save('natural-watch-private.json',{episode:e,initial:t,newServicePid:secondPid,startedAt:new Date().toISOString()});console.log(JSON.stringify({naturalWaitStarted:true,claimedAt:t.claimedAt,taskId:t.id,newServicePid:secondPid,minimumExpiryAt:new Date(Date.parse(t.claimedAt)+1800000).toISOString()}));
