// @vitest-environment node
import {expect,it} from 'vitest';
import {readFile,writeFile,mkdir} from 'node:fs/promises';
import {join} from 'node:path';
import {createHash} from 'node:crypto';
import {formalWorkflowFixture} from './formalWorkflow.fixture';
import {historicalProductionScriptV2} from '../reviews/productionScript.fixture';
import {defaultShotCaptionContract} from '../shotCaptions';
import {workspaceTestMedia} from './workspaceTestMedia.fixture';

const mixedScript=`# 混合制作输入

## 制作设置
- 模板版本：3
- 画幅：9:16
- 字幕：关闭

## 正文
### 段落 01
- 口播：图片镜头保留自己的正文。
- 画面来源：图片
- 素材名称：still.png
- 镜头时长（秒）：2

### 段落 02
- 口播：已有素材保留独立原声。
- 画面来源：已有素材
- 素材名称：source.mp4
- 镜头时长（秒）：2

### 段落 03
- 口播：尚未选择画面不产生生成任务。
- 画面来源：暂未选择
- 镜头时长（秒）：2

### 段落 04
- 口播：视频意图仍需明确提交。
- 画面来源：视频
- 首帧素材：still.png
- 视频生成提示词：固定机位，保留主体。
- 生成时长（秒）：4
`;

async function noRetiredWrites(f:Awaited<ReturnType<typeof formalWorkflowFixture>>){
 for(const table of ['public.review_packages','public.shot_preparation_drafts','public.approvals','local_identity.workspace_shot_inputs']){
  expect((await f.c.query(`SELECT count(*)::int n FROM ${table} WHERE episode_id=$1`,[f.newEpisode])).rows[0].n,table).toBe(0);
 }
}

it.skipIf(process.env.LOOP_WORKSPACE_ACCEPTANCE!=='1').each([
 ['prose','# 普通正文\n\n## 正文\n\n第一镜头有自己的字幕。\n\n第二镜头保留自己的口播。',2],
 ['v2',historicalProductionScriptV2,1],
 ['v3-image',mixedScript.slice(0,mixedScript.indexOf('### 段落 02')),1],
 ['v3-mixed',mixedScript,4],
] as const)('B1 remaining %s: independent captions, manual voice and matched media survive format continuation without retired writes',async(kind,text,count)=>{
 const f=await formalWorkflowFixture({freshAccount:true,scriptText:text});
 try{
  expect((await f.send('start')).status).toBe(200);
  const made=await f.send('prepareWorkspace',{expectedScriptRevisionId:f.main});expect(made.status,JSON.stringify(made.body)).toBe(200);let w=made.body;
  expect(w.clips).toHaveLength(count);expect(w.storyboardPackageId).toBeNull();await noRetiredWrites(f);
  const stable=w.clips.map((c:any)=>({id:c.id,itemId:c.itemId,shotId:c.shotId}));
  const originalBaseline=(await f.c.query('SELECT * FROM local_identity.production_preparation_revisions WHERE id=$1',[w.preparationRevisionId])).rows[0];
  const originalBytes=await readFile(join(f.assetRoot,originalBaseline.file_reference.relativePath));
  expect(createHash('sha256').update(originalBytes).digest('hex')).toBe(originalBaseline.file_reference.sha256);
  for(const shot of originalBaseline.manifest.shots)expect(shot).not.toHaveProperty('shotType');
  const send=async(command:string,data:Record<string,unknown>={})=>{const r=await f.send(command,{workspaceId:w.id,expectedRevision:w.revision,...data});expect(r.status,command+': '+JSON.stringify(r.body)).toBe(200);if(r.body.id===w.id)w=r.body;return r.body;};
  const upload=async(name:string,type:string,mime:string,purpose:string)=>{const r=await f.send('importMaterial',{sourceKind:'file',sourcePath:name,materialType:type,materialPurpose:purpose,mimeType:mime,isMainScript:false,contentBase64:(await readFile(workspaceTestMedia(name==='still.png'?'still-striped.png':name==='voice.mp3'?'controlled-narration-tone.mp3':'landscape-red-blue.mp4'))).toString('base64')});expect(r.status,JSON.stringify(r.body)).toBe(200);return r.body.id;};
  const still=await upload('still.png','image','image/png','visual_reference');
  await send('matchWorkspaceImage',{clipId:w.clips[0].id,materialRevisionId:still,usage:'still',durationSeconds:2,fit:'contain',focusX:0,focusY:0});
  if(kind==='v3-mixed'){
   const video=await upload('source.mp4','video','video/mp4','a_roll');
   await send('matchWorkspaceMaterial',{clipId:w.clips[1].id,materialRevisionId:video});
   expect(w.readiness.placeholders).toContain(w.clips[2].itemId);expect(w.readiness.placeholders).toContain(w.clips[3].itemId);
  }
  for(const [index,clip] of w.clips.entries()){
   const contract=defaultShotCaptionContract(`人工字幕 ${kind} ${index+1}`,'independent');contract.cues=[{id:'manual-'+index,text:contract.text,startMs:0,endMs:900}];
   await send('saveWorkspaceCaptions',{clipId:clip.id,captionContract:contract});
   await send('saveWorkspaceNarration',{clipId:clip.id,mode:'none',text:`人工口播输入 ${kind} ${index+1}`});
  }
  const voice=await upload('voice.mp3','audio','audio/mpeg','general_reference');
  await send('saveWorkspaceNarration',{clipId:w.clips[0].id,mode:'manual',text:'人工录音正文保持。',materialRevisionId:voice});
  const track=w.narration.find((n:any)=>n.shotId===w.clips[0].shotId).audioTrackId;
  expect(track).toBeTruthy();expect(w.usedNarration).toEqual([]);
  const prior={project:w.project,clips:w.clips,captions:w.captions.map((c:any)=>c.contract),narration:w.narration.map((n:any)=>({shotId:n.shotId,text:n.text,mode:n.mode,audioTrackId:n.audioTrackId})),basis:w.preparationRevisionId};
  const snapshot=(await f.c.query('SELECT input_snapshot FROM public.tasks WHERE id=(SELECT source_task_id FROM public.audio_tracks WHERE id=$1)',[track])).rows[0].input_snapshot;
  expect(snapshot.workspace_narration).toMatchObject({version:'workspace-narration/v2',workspace_id:w.id,preparation_revision_id:prior.basis,shot_id:w.clips[0].shotId});
  expect(snapshot).not.toHaveProperty('shot_preparation');expect(snapshot).not.toHaveProperty('storyboard_review_package_id');
  const revision=await f.send('reconfirmScript',{expectedScriptRevisionId:f.main,text:text.replace('## 正文','\n## 正文'),reason:'只增加空行，保留所有人工输入'});expect(revision.status,JSON.stringify(revision.body)).toBe(200);
  await send('resumeWorkspaceEditing',{expectedScriptRevisionId:revision.body.id,expectedProjectSha256:w.project.sha256});
  expect(w.preparationRevisionId).not.toBe(prior.basis);expect(w.project).toEqual(prior.project);expect(w.clips).toEqual(prior.clips);
  expect(w.clips.map((c:any)=>({id:c.id,itemId:c.itemId,shotId:c.shotId}))).toEqual(stable);
  expect(w.captions.map((c:any)=>c.contract)).toEqual(prior.captions);
  expect(w.narration.map((n:any)=>({shotId:n.shotId,text:n.text,mode:n.mode,audioTrackId:n.audioTrackId}))).toEqual(prior.narration);
  expect((await f.c.query('SELECT count(*)::int n FROM local_identity.production_shots WHERE episode_id=$1',[f.newEpisode])).rows[0].n).toBe(count);
  expect((await f.c.query('SELECT * FROM local_identity.production_preparation_revisions WHERE id=$1',[prior.basis])).rows[0]).toEqual(originalBaseline);
  expect(await readFile(join(f.assetRoot,originalBaseline.file_reference.relativePath))).toEqual(originalBytes);
  expect((await f.c.query('SELECT preparation_revision_id,source_review_package_id FROM public.audio_tracks WHERE id=$1',[track])).rows[0]).toEqual({preparation_revision_id:prior.basis,source_review_package_id:null});
  expect((await f.c.query('SELECT input_snapshot FROM public.tasks WHERE id=(SELECT source_task_id FROM public.audio_tracks WHERE id=$1)',[track])).rows[0].input_snapshot).toEqual(snapshot);
  await f.restart();const response=await fetch(`${f.origin}/api/local/episodes/${f.newEpisode}/workspace`,{headers:{Cookie:'loop_owner_session='+f.token}});expect(response.status).toBe(200);const reopened=(await response.json()).workspace;
  expect(reopened.project).toEqual(w.project);expect(reopened.clips).toEqual(w.clips);expect(reopened.captions.map((c:any)=>c.contract)).toEqual(prior.captions);expect(reopened.narration.map((n:any)=>({shotId:n.shotId,text:n.text,mode:n.mode,audioTrackId:n.audioTrackId}))).toEqual(prior.narration);
  await noRetiredWrites(f);
  const tasks=(await f.c.query('SELECT task_type FROM public.tasks WHERE episode_id=$1',[f.newEpisode])).rows;expect(tasks).toEqual([{task_type:'generate_narration'}]);
  expect((await f.c.query('SELECT count(*)::int n FROM public.paid_generation_reservations r JOIN public.tasks t ON t.id=r.task_id WHERE t.episode_id=$1',[f.newEpisode])).rows[0].n).toBe(0);
  if(process.env.LOOP_STORY_EVIDENCE){await mkdir(process.env.LOOP_STORY_EVIDENCE,{recursive:true});await writeFile(join(process.env.LOOP_STORY_EVIDENCE,`b1-input-${kind}.json`),JSON.stringify({kind,episode:f.newEpisode,baselineBefore:prior.basis,baselineAfter:w.preparationRevisionId,stable,manualTrack:track,project:w.project,oldTablesEmpty:true,tasks},null,2));}
 }finally{await f.close();}
},180000);
